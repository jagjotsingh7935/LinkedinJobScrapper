import React, { useState } from 'react';
import {
  Box, Typography, TextField, Button, CircularProgress, Alert, Card, CardContent, 
  Grid, Chip, Container, Paper, InputAdornment, Divider, Stack, IconButton,
  useTheme, useMediaQuery, Fade, Skeleton, alpha
} from '@mui/material';
import {
  Search as SearchIcon,
  LocationOn as LocationIcon,
  Work as WorkIcon,
  Download as DownloadIcon,
  Business as BusinessIcon,
  OpenInNew as OpenInNewIcon,
  FilterList as FilterIcon,
  TrendingUp as TrendingUpIcon,
  Speed as SpeedIcon
} from '@mui/icons-material';
import { searchJobs, downloadExcel } from './api';

function JobSearch() {
  const [keywords, setKeywords] = useState('');
  const [location, setLocation] = useState('');
  const [jobLimit, setJobLimit] = useState(100);
  const [jobs, setJobs] = useState([]);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState(null);
  const [message, setMessage] = useState('');
  const [downloadLoading, setDownloadLoading] = useState(false);

  const theme = useTheme();
  const isMobile = useMediaQuery(theme.breakpoints.down('sm'));
  const isTablet = useMediaQuery(theme.breakpoints.down('md'));

  const handleSearch = async () => {
    setLoading(true);
    setError(null);
    setMessage('');
    try {
      const data = await searchJobs(keywords, location, jobLimit);
      setJobs(data.jobs || []);
      setMessage(data.message || '');
    } catch (err) {
      setError('Failed to fetch jobs. Please try again.');
    } finally {
      setLoading(false);
    }
  };

  const handleDownload = async () => {
    setDownloadLoading(true);
    try {
      await downloadExcel(jobs);
    } catch (err) {
      setError('Failed to download Excel file.');
    } finally {
      setDownloadLoading(false);
    }
  };

  const handleKeyPress = (e) => {
    if (e.key === 'Enter' && keywords && location) {
      handleSearch();
    }
  };

  const renderJobCard = (job) => (
    <Grid item xs={12} sm={6} lg={4} key={job.job_id}>
      <Card 
        elevation={0}
        sx={{
          height: '100%',
          display: 'flex',
          flexDirection: 'column',
          transition: 'all 0.3s ease-in-out',
          background: `linear-gradient(145deg, ${theme.palette.grey[800]} 0%, ${theme.palette.grey[900]} 100%)`,
          color: theme.palette.common.white,
          borderRadius: 3,
          border: `1px solid ${alpha(theme.palette.primary.main, 0.2)}`,
          position: 'relative',
          overflow: 'hidden',
          '&:hover': {
            transform: 'translateY(-6px)',
            boxShadow: `0 8px 32px ${alpha(theme.palette.primary.main, 0.3)}`,
            border: `1px solid ${alpha(theme.palette.primary.main, 0.4)}`,
          },
          '&::before': {
            content: '""',
            position: 'absolute',
            top: 0,
            left: 0,
            right: 0,
            height: 3,
            background: `linear-gradient(90deg, ${theme.palette.primary.main} 0%, ${theme.palette.primary.dark} 100%)`,
          },
        }}
      >
        <CardContent sx={{ flexGrow: 1, p: 3 }}>
          <Box sx={{ display: 'flex', alignItems: 'flex-start', mb: 2 }}>
            <Box sx={{
              p: 1,
              borderRadius: 2,
              background: `linear-gradient(135deg, ${theme.palette.primary.main} 0%, ${theme.palette.primary.dark} 100%)`,
              mr: 2,
              mt: 0.5,
              boxShadow: `0 4px 12px ${alpha(theme.palette.primary.main, 0.3)}`,
            }}>
              <BusinessIcon sx={{ color: 'white', fontSize: 20 }} />
            </Box>
            <Box sx={{ flexGrow: 1 }}>
              <Typography 
                variant="h6" 
                component="h3"
                sx={{ 
                  fontWeight: 600,
                  mb: 0.5,
                  lineHeight: 1.3,
                  color: 'white'
                }}
              >
                {job.job_title}
              </Typography>
              <Typography 
                variant="body2" 
                sx={{ 
                  color: theme.palette.primary.light,
                  fontWeight: 500,
                  mb: 0.5
                }}
              >
                {job.company}
              </Typography>
              <Box sx={{ display: 'flex', alignItems: 'center', mb: 1 }}>
                <LocationIcon sx={{ fontSize: 16, color: alpha(theme.palette.common.white, 0.7), mr: 0.5 }} />
                <Typography variant="body2" sx={{ color: alpha(theme.palette.common.white, 0.7) }}>
                  {job.location}
                </Typography>
              </Box>
            </Box>
          </Box>

          <Typography 
            variant="body2" 
            sx={{ 
              mb: 2, 
              color: alpha(theme.palette.common.white, 0.8),
              lineHeight: 1.4,
              display: '-webkit-box',
              WebkitBoxOrient: 'vertical',
              WebkitLineClamp: 3,
              overflow: 'hidden'
            }}
          >
            {job.job_description?.substring(0, 120)}...
          </Typography>

          {job.skills && job.skills.length > 0 && (
            <Box sx={{ mb: 2 }}>
              <Typography variant="caption" sx={{ 
                mb: 1, 
                display: 'block',
                color: alpha(theme.palette.common.white, 0.6),
                fontWeight: 500,
                textTransform: 'uppercase',
                letterSpacing: '0.5px'
              }}>
                Skills Required:
              </Typography>
              <Box sx={{ display: 'flex', flexWrap: 'wrap', gap: 0.5 }}>
                {job.skills.slice(0, 4).map((skill, index) => (
                  <Chip 
                    key={index} 
                    label={skill} 
                    size="small"
                    sx={{
                      backgroundColor: alpha(theme.palette.primary.main, 0.2),
                      color: theme.palette.primary.light,
                      fontSize: '0.75rem',
                      height: 24,
                      border: `1px solid ${alpha(theme.palette.primary.main, 0.3)}`,
                      '&:hover': {
                        backgroundColor: alpha(theme.palette.primary.main, 0.3),
                      }
                    }}
                  />
                ))}
                {job.skills.length > 4 && (
                  <Chip 
                    label={`+${job.skills.length - 4} more`}
                    size="small"
                    sx={{
                      backgroundColor: alpha(theme.palette.common.white, 0.1),
                      color: alpha(theme.palette.common.white, 0.7),
                      fontSize: '0.75rem',
                      height: 24,
                      border: `1px solid ${alpha(theme.palette.common.white, 0.2)}`,
                    }}
                  />
                )}
              </Box>
            </Box>
          )}

          <Button
            href={job.job_url}
            target="_blank"
            variant="contained"
            size="small"
            endIcon={<OpenInNewIcon />}
            sx={{ 
              mt: 'auto',
              borderRadius: 2,
              textTransform: 'none',
              fontWeight: 600,
              background: `linear-gradient(135deg, ${theme.palette.primary.main} 0%, ${theme.palette.primary.dark} 100%)`,
              boxShadow: `0 4px 12px ${alpha(theme.palette.primary.main, 0.3)}`,
              '&:hover': {
                background: `linear-gradient(135deg, ${theme.palette.primary.dark} 0%, ${theme.palette.primary.main} 100%)`,
                boxShadow: `0 6px 20px ${alpha(theme.palette.primary.main, 0.4)}`,
                transform: 'translateY(-2px)',
              }
            }}
          >
            View Details
          </Button>
        </CardContent>
      </Card>
    </Grid>
  );

  const renderLoadingSkeletons = () => (
    Array.from({ length: 6 }).map((_, index) => (
      <Grid item xs={12} sm={6} lg={4} key={index}>
        <Card 
          elevation={0}
          sx={{ 
            height: 300,
            background: `linear-gradient(145deg, ${theme.palette.grey[800]} 0%, ${theme.palette.grey[900]} 100%)`,
            borderRadius: 3,
            border: `1px solid ${alpha(theme.palette.primary.main, 0.2)}`,
          }}
        >
          <CardContent sx={{ p: 3 }}>
            <Skeleton variant="text" width="80%" height={32} sx={{ bgcolor: alpha(theme.palette.common.white, 0.1) }} />
            <Skeleton variant="text" width="60%" height={24} sx={{ mb: 1, bgcolor: alpha(theme.palette.common.white, 0.1) }} />
            <Skeleton variant="text" width="50%" height={20} sx={{ mb: 2, bgcolor: alpha(theme.palette.common.white, 0.1) }} />
            <Skeleton variant="rectangular" width="100%" height={60} sx={{ mb: 2, bgcolor: alpha(theme.palette.common.white, 0.1) }} />
            <Box sx={{ display: 'flex', gap: 1, mb: 2 }}>
              <Skeleton variant="rounded" width={60} height={24} sx={{ bgcolor: alpha(theme.palette.common.white, 0.1) }} />
              <Skeleton variant="rounded" width={80} height={24} sx={{ bgcolor: alpha(theme.palette.common.white, 0.1) }} />
            </Box>
            <Skeleton variant="rounded" width={100} height={32} sx={{ bgcolor: alpha(theme.palette.common.white, 0.1) }} />
          </CardContent>
        </Card>
      </Grid>
    ))
  );

  return (
    <Box
      sx={{
        minHeight: '100vh',
        width: '100%',
        background: `linear-gradient(145deg, ${theme.palette.grey[900]} 0%, ${theme.palette.grey[800]} 100%)`,
        color: theme.palette.common.white,
        display: 'flex',
        flexDirection: 'column',
        padding: { xs: 1, sm: 2, md: 3 },
      }}
    >
      {/* Hero Section */}
      <Paper 
        elevation={0} 
        sx={{ 
          p: { xs: 2, sm: 3, md: 4 },
          mb: 4,
          background: `linear-gradient(145deg, ${theme.palette.grey[900]} 0%, ${theme.palette.grey[800]} 100%)`,
          borderRadius: 2,
          color: 'white',
          position: 'relative',
          overflow: 'hidden',
          boxShadow: theme.shadows[8],
          '&::before': {
            content: '""',
            position: 'absolute',
            top: 0,
            left: 0,
            right: 0,
            bottom: 0,
            background: `radial-gradient(circle at 20% 50%, ${alpha(theme.palette.primary.main, 0.2)} 0%, transparent 50%)`,
            pointerEvents: 'none',
          },
        }}
      >
        <Box sx={{ textAlign: 'center', mb: 4, position: 'relative', zIndex: 1 }}>
          <Box sx={{
            display: 'inline-flex',
            p: 1.5,
            borderRadius: 2,
            background: `linear-gradient(135deg, ${theme.palette.primary.main} 0%, ${theme.palette.primary.dark} 100%)`,
            mb: 2,
            boxShadow: `0 6px 18px ${alpha(theme.palette.primary.main, 0.3)}`,
          }}>
            <WorkIcon sx={{ fontSize: 30, color: 'white' }} />
          </Box>
          <Typography 
            variant="h3" 
            component="h1"
            sx={{ 
              fontWeight: 700,
              mb: 1.5,
              fontSize: { xs: '1.8rem', sm: '2.2rem', md: '2.8rem' },
              background: `linear-gradient(135deg, ${theme.palette.common.white} 0%, ${theme.palette.primary.light} 100%)`,
              backgroundClip: 'text',
              WebkitBackgroundClip: 'text',
              WebkitTextFillColor: 'transparent',
            }}
          >
            Find Your Dream Job
          </Typography>
          <Typography variant="h6" sx={{ 
            opacity: 0.9, 
            maxWidth: 600, 
            mx: 'auto',
            color: alpha(theme.palette.common.white, 0.8),
            fontSize: { xs: '0.9rem', sm: '1rem', md: '1.1rem' }
          }}>
            Discover opportunities that match your skills and aspirations
          </Typography>
        </Box>

        {/* Search Form */}
        <Paper 
          elevation={0}
          sx={{ 
            p: { xs: 2, sm: 3 },
            borderRadius: 2,
            background: `linear-gradient(145deg, ${alpha(theme.palette.grey[800], 0.8)} 0%, ${alpha(theme.palette.grey[900], 0.8)} 100%)`,
            backdropFilter: 'blur(20px)',
            border: `1px solid ${alpha(theme.palette.primary.main, 0.2)}`,
            position: 'relative',
            zIndex: 1,
          }}
        >
          <Stack 
            direction={{ xs: 'column', sm: 'row' }} 
            spacing={2}
            sx={{ mb: 2 }}
          >
            <TextField
              label="Job Keywords"
              placeholder="e.g., Software Engineer, Data Scientist"
              value={keywords}
              onChange={(e) => setKeywords(e.target.value)}
              onKeyPress={handleKeyPress}
              fullWidth
              InputProps={{
                startAdornment: (
                  <InputAdornment position="start">
                    <SearchIcon sx={{ color: theme.palette.primary.main }} />
                  </InputAdornment>
                ),
              }}
              sx={{
                '& .MuiOutlinedInput-root': {
                  borderRadius: 1.5,
                  backgroundColor: alpha(theme.palette.common.white, 0.05),
                  '& fieldset': {
                    borderColor: alpha(theme.palette.common.white, 0.2),
                  },
                  '&:hover fieldset': {
                    borderColor: alpha(theme.palette.primary.main, 0.5),
                  },
                  '&.Mui-focused fieldset': {
                    borderColor: theme.palette.primary.main,
                  },
                },
                '& .MuiInputLabel-root': {
                  color: alpha(theme.palette.common.white, 0.7),
                },
                '& .MuiInputBase-input': {
                  color: theme.palette.common.white,
                },
              }}
            />
            <TextField
              label="Location"
              placeholder="e.g., New York, Remote, London"
              value={location}
              onChange={(e) => setLocation(e.target.value)}
              onKeyPress={handleKeyPress}
              fullWidth
              InputProps={{
                startAdornment: (
                  <InputAdornment position="start">
                    <LocationIcon sx={{ color: theme.palette.primary.main }} />
                  </InputAdornment>
                ),
              }}
              sx={{
                '& .MuiOutlinedInput-root': {
                  borderRadius: 1.5,
                  backgroundColor: alpha(theme.palette.common.white, 0.05),
                  '& fieldset': {
                    borderColor: alpha(theme.palette.common.white, 0.2),
                  },
                  '&:hover fieldset': {
                    borderColor: alpha(theme.palette.primary.main, 0.5),
                  },
                  '&.Mui-focused fieldset': {
                    borderColor: theme.palette.primary.main,
                  },
                },
                '& .MuiInputLabel-root': {
                  color: alpha(theme.palette.common.white, 0.7),
                },
                '& .MuiInputBase-input': {
                  color: theme.palette.common.white,
                },
              }}
            />
            <TextField
              label="Job Limit"
              type="number"
              value={jobLimit}
              onChange={(e) => setJobLimit(e.target.value)}
              sx={{
                minWidth: { xs: '100%', sm: '120px' },
                '& .MuiOutlinedInput-root': {
                  borderRadius: 1.5,
                  backgroundColor: alpha(theme.palette.common.white, 0.05),
                  '& fieldset': {
                    borderColor: alpha(theme.palette.common.white, 0.2),
                  },
                  '&:hover fieldset': {
                    borderColor: alpha(theme.palette.primary.main, 0.5),
                  },
                  '&.Mui-focused fieldset': {
                    borderColor: theme.palette.primary.main,
                  },
                },
                '& .MuiInputLabel-root': {
                  color: alpha(theme.palette.common.white, 0.7),
                },
                '& .MuiInputBase-input': {
                  color: theme.palette.common.white,
                },
              }}
              InputProps={{
                startAdornment: (
                  <InputAdornment position="start">
                    <SpeedIcon sx={{ color: theme.palette.primary.main }} />
                  </InputAdornment>
                ),
              }}
            />
          </Stack>

          <Box sx={{ display: 'flex', justifyContent: 'center' }}>
            <Button
              variant="contained"
              onClick={handleSearch}
              disabled={loading || !keywords || !location}
              startIcon={loading ? <CircularProgress size={20} /> : <SearchIcon />}
              size="large"
              sx={{ 
                minWidth: { xs: '100%', sm: '200px' },
                py: 1,
                borderRadius: 1.5,
                textTransform: 'none',
                fontWeight: 600,
                fontSize: '1rem',
                background: `linear-gradient(135deg, ${theme.palette.primary.main} 0%, ${theme.palette.primary.dark} 100%)`,
                boxShadow: `0 6px 20px ${alpha(theme.palette.primary.main, 0.4)}`,
                '&:hover': {
                  background: `linear-gradient(135deg, ${theme.palette.primary.dark} 0%, ${theme.palette.primary.main} 100%)`,
                  boxShadow: `0 8px 28px ${alpha(theme.palette.primary.main, 0.5)}`,
                  transform: 'translateY(-2px)',
                },
                '&:disabled': {
                  background: alpha(theme.palette.common.white, 0.1),
                  color: alpha(theme.palette.common.white, 0.5),
                }
              }}
            >
              {loading ? 'Searching...' : 'Search Jobs'}
            </Button>
          </Box>
        </Paper>
      </Paper>

      {/* Error Alert */}
      {error && (
        <Fade in={!!error}>
          <Alert 
            severity="error" 
            sx={{ 
              mb: 2,
              borderRadius: 1.5,
              backgroundColor: alpha(theme.palette.error.main, 0.1),
              color: theme.palette.error.main,
              border: `1px solid ${alpha(theme.palette.error.main, 0.2)}`,
              '& .MuiAlert-icon': {
                alignItems: 'center',
                color: theme.palette.error.main,
              }
            }}
            onClose={() => setError(null)}
          >
            {error}
          </Alert>
        </Fade>
      )}

      {/* Info Alert */}
      {message && (
        <Fade in={!!message}>
          <Alert 
            severity="info" 
            sx={{ 
              mb: 2,
              borderRadius: 1.5,
              backgroundColor: alpha(theme.palette.info.main, 0.1),
              color: theme.palette.info.main,
              border: `1px solid ${alpha(theme.palette.info.main, 0.2)}`,
              '& .MuiAlert-icon': {
                alignItems: 'center',
                color: theme.palette.info.main,
              }
            }}
            onClose={() => setMessage('')}
          >
            {message}
          </Alert>
        </Fade>
      )}

      {/* Results Section */}
      {(jobs.length > 0 || loading) && (
        <Box sx={{ flexGrow: 1 }}>
          <Paper 
            elevation={0}
            sx={{ 
              p: 2,
              mb: 2,
              borderRadius: 2,
              background: `linear-gradient(145deg, ${theme.palette.grey[800]} 0%, ${theme.palette.grey[900]} 100%)`,
              border: `1px solid ${alpha(theme.palette.primary.main, 0.2)}`,
              color: theme.palette.common.white,
            }}
          >
            <Box sx={{ 
              display: 'flex', 
              justifyContent: 'space-between',
              alignItems: 'center',
              flexDirection: { xs: 'column', sm: 'row' },
              gap: 1.5
            }}>
              <Box sx={{ display: 'flex', alignItems: 'center', gap: 1 }}>
                <Box sx={{
                  p: 1,
                  borderRadius: 1.5,
                  background: `linear-gradient(135deg, ${theme.palette.primary.main} 0%, ${theme.palette.primary.dark} 100%)`,
                  boxShadow: `0 4px 12px ${alpha(theme.palette.primary.main, 0.3)}`,
                }}>
                  <TrendingUpIcon sx={{ color: 'white' }} />
                </Box>
                <Typography variant="h5" sx={{ fontWeight: 600, color: 'white' }}>
                  {loading ? 'Searching...' : `${jobs.length} Jobs Found`}
                </Typography>
              </Box>
              
              {jobs.length > 0 && (
                <Button
                  variant="outlined"
                  onClick={handleDownload}
                  disabled={downloadLoading}
                  startIcon={downloadLoading ? <CircularProgress size={20} /> : <DownloadIcon />}
                  sx={{
                    borderRadius: 1.5,
                    textTransform: 'none',
                    fontWeight: 500,
                    minWidth: { xs: '100%', sm: 'auto' },
                    borderColor: alpha(theme.palette.primary.main, 0.5),
                    color: theme.palette.primary.light,
                    '&:hover': {
                      borderColor: theme.palette.primary.main,
                      backgroundColor: alpha(theme.palette.primary.main, 0.1),
                    }
                  }}
                >
                  {downloadLoading ? 'Downloading...' : 'Download Excel'}
                </Button>
              )}
            </Box>
          </Paper>

          <Grid container spacing={2}>
            {loading ? renderLoadingSkeletons() : jobs.map(renderJobCard)}
          </Grid>
        </Box>
      )}

      {/* Empty State */}
      {!loading && jobs.length === 0 && (keywords || location) && (
        <Paper 
          elevation={0}
          sx={{ 
            p: { xs: 2, sm: 4 },
            textAlign: 'center',
            borderRadius: 2,
            background: `linear-gradient(145deg, ${theme.palette.grey[800]} 0%, ${theme.palette.grey[900]} 100%)`,
            border: `1px solid ${alpha(theme.palette.primary.main, 0.2)}`,
            color: theme.palette.common.white,
          }}
        >
          <Box sx={{
            display: 'inline-flex',
            p: 2,
            borderRadius: 2,
            background: alpha(theme.palette.common.white, 0.05),
            mb: 2,
          }}>
            <SearchIcon sx={{ fontSize: 40, color: alpha(theme.palette.common.white, 0.5) }} />
          </Box>
          <Typography variant="h6" sx={{ mb: 1, color: 'white' }}>
            No jobs found
          </Typography>
          <Typography variant="body2" sx={{ color: alpha(theme.palette.common.white, 0.7) }}>
            Try adjusting your search criteria or keywords
          </Typography>
        </Paper>
      )}
    </Box>
  );
}

export default JobSearch;
